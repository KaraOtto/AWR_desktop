JAP_nf_sp_shusui = {
	specialization = specialization_air
	icon = GFX_sp_rockets_rocket_interceptor
	project_tags = { sp_tag_rocket sp_tag_aircraft }
	allowed = {
		original_tag = JAP
		has_dlc = "Gotterdammerung"
	}
	visible = {
		FROM = {
			has_completed_focus = qsjh
		}
	}
	available = {
		FROM = {
			has_tech = experimental_rockets
		}
	}
	breakthrough_cost = {
		specialization_air = 1
	}
	resource_cost = {
		resources = {
			aluminium = 8
			tungsten = 4
			chromium = 4
		}
	}
	prototype_time = sp_time.prototype.medium
	complexity = sp_complexity.medium
	ai_will_do = {
		base = 1
	}
	project_output = {
		country_effects = {
			if = {
				limit = {
					NOT = {
						is_special_project_completed = sp:sp_rocket_interceptor
					}
				}
				custom_effect_tooltip = {
					localization_key = SP_UNLOCK_PROJECT

				}
				hidden_effect = {
					complete_special_project = sp:sp_rocket_interceptor
				}
			}
			JAP_nf_add_or_modify_air = yes
			add_to_variable = {
				JAP_nf_air_air_superiority_efficiency = 0.05
				tooltip = air_superiority_efficiency_tt
			}
		}
	}
	generic_prototype_rewards = {
		sp_rockets_scientist_xp_generic_reward
		sp_rockets_political_interference_generic_reward
		sp_rockets_onsite_explosion_generic_reward

		sp_air_generic_reward_scientist_xp_1
		sp_air_generic_reward_scientist_xp_2
		sp_air_generic_reward_air_xp_1
		sp_air_generic_reward_air_xp_2

		sp_air_generic_reward_major_progress_2
		sp_air_generic_reward_test_failure_1
		sp_air_generic_reward_test_failure_2
	}
	unique_prototype_rewards = {
		JAP_nf_sp_shusui_fuel_reward = {
			fire_only_once = yes
			threshold = {
				min = 20
				max = 80
			}
			weight = {
				base = 100
			}
			option = {
				token = JAP_nf_sp_shusui_fuel_safe
				default = yes
				iteration_output = {
					country_effects = {
						FROM = {
							add_project_progress_ratio = constant:sp_progress.gain.low
						}
					}

						add_scientist_xp = {
							experience = constant:sp_scientist_xp_gain.medium

						}
					}
				}
			}
			option = {
				token = JAP_nf_sp_shusui_fuel_rush

						FROM = {
