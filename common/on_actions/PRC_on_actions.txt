	# 避免：

	# 数组中的目标已经过期

				}

				PRC_setup_third_world_independence_targets = yes
			}
		}
	}

	# 每月检查所有国家
	# on_monthly本身会针对每个国家执行
	# ROOT = 当前被检查国家
	on_monthly = {

		effect = {

			if = {

				limit = {
					# 排除PRC自己
					NOT = {
						original_tag = PRC
					}
					# 国家必须已经真正独立
					is_subject = no
					# 独立战争必须已经结束
					has_war = no

					NOT = {
						has_country_flag = PRC_third_world_join_request_sent
					}

					# 该民族解放运动

						# 原来就是一个殖民附属国
						has_country_flag = PRC_supported_independence_movement
						# 原来不存在
						# 是通过殖民地革命重新出现

							has_state_flag = PRC_supported_revolution_state
							is_core_of = ROOT
						}
					}
					# PRC自身必须存在
					# 并且是阵营领袖
					PRC = {

						exists = yes

						has_country_flag = PRC_third_world_support_unlocked

						
                        OR = {
                            AND = {
                                has_completed_focus = PRC_internationalist_clique_focus
                                is_faction_leader = yes
                            }
                            AND = {
                                has_completed_focus = PRC_support_the_national_independence_movement

						NOT = {
							has_war_with = ROOT
						}
					}
				}

				# 已经申请过一次
				set_country_flag = PRC_third_world_join_request_sent

                if = {

                    hours = 6
                }
