on_actions = {	#民国权力集团
	on_startup = {
		effect = {
			CHI = {
				if = {
					limit = {
						NOT = {
							has_country_flag = CHI_power_blocs_initialized
						}
					}
					CHI_power_blocs_initialize = yes
				}
				else = {
					CHI_power_blocs_normalize_and_refresh = yes
				}
				if = {
					limit = {
						NOT = {
							has_country_flag = CHI_corruption_initialized
						}
					}
					CHI_corruption_initialize = yes
				}

				country_event = {
					id = AWR_CHI.4
					days = 14
				}
				country_event = {
					id = AWR_CHI.6
					days = 3
				}
			}
		}
	}

	on_monthly_CHI = {
		effect = {
			if = {
				limit = {
					has_country_flag = CHI_power_blocs_initialized
				}
				CHI_power_blocs_normalize_and_refresh = yes
			}
		}
	}
	# 兼容已开始的旧存档：在下一个游戏日补齐开局状态。
	on_daily_CHI = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_country_flag = CHI_power_blocs_initialized
					}
				}
				CHI_power_blocs_initialize = yes
			}
		}
	}
}

on_actions = {	#腐败
	on_startup = {
		effect = {
			CHI = {
				if = {
					limit = {
						NOT = {
							has_country_flag = CHI_corruption_initialized
						}
					}
					CHI_corruption_initialize = yes
				}
				CHI_corruption_reform_migrate = yes
				CHI_corruption_monthly_update = yes
			}
		}
	}

	on_monthly_CHI = {
		effect = {
			if = {
				limit = {
					NOT = {
						has_country_flag = CHI_corruption_initialized
					}
				}
				CHI_corruption_initialize = yes
			}
		}
	}

	on_daily_CHI = {
		effect = {
			if = {
				limit = {
					has_country_flag = CHI_corruption_initialized
					NOT = {
						has_country_flag = CHI_corruption_abolished
					}
				}

				CHI_corruption_reform_migrate = yes
				CHI_corruption_monthly_update = yes
			}
		}
	}

	on_monthly_CHI = {
		effect = {
			if = {
				limit = {
					has_country_flag = CHI_corruption_initialized
					NOT = {
						has_country_flag = CHI_corruption_abolished
					}
				}

				CHI_corruption_monthly_tick = yes
			}
		}
	}
}
