			ratio > 0.54
			sub_unit_def_weights = { carrier = 2 battleship = 1 battle_cruiser = 1 heavy_cruiser = 0.5 light_cruiser = 0.25 destroyer = 0.1 submarine = 0.05 }
		}
	}
}
