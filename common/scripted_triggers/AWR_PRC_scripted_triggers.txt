PRC_is_valid_liberation_target = {

	# 中国自己永远不能成为目标
	NOT = {
		original_tag = PRC
	}

	OR = {
		################################
		# A. 已经存在的殖民附属国
		################################
		AND = {

			exists = yes

			is_subject = yes

			# 中国自己的附属国不需要“解放”
			NOT = {
				is_subject_of = PRC
			}
		}
		################################
		# B. 尚未存在的殖民地民族国家
		################################
		AND = {

			exists = no

			# 至少还有一个核心州
			# 正处于外国的非核心统治下
			any_core_state = {

				NOT = {
					is_core_of = owner
				}

				owner = {
					NOT = {
						original_tag = PRC
					}
				}
			}
		}
	}
}
