AWR_apply_country_limits = {
	if = {
		limit = {
			has_game_rule = {
				rule = AWR_country_limits
				option = LIMIT_900
			}
		}
		set_variable = {
			AI_army_num_divisions = 900
		}
		set_variable = {
			AI_aircraft_num_divisions = 20000
		}
	}
	if = {
		limit = {
			has_game_rule = {
				rule = AWR_country_limits
				option = LIMIT_600
			}
		}
		set_variable = {
			AI_army_num_divisions = 600
		}
		set_variable = {
			AI_aircraft_num_divisions = 10000
		}
	}
	if = {
		limit = {
			has_game_rule = {
				rule = AWR_country_limits
				option = LIMIT_300
			}
		}
		set_variable = {
			AI_army_num_divisions = 300
		}
		set_variable = {
			AI_aircraft_num_divisions = 5000
		}
	}
}

AWR_show_country_startup_advice = {

		limit = {
			tag = PRC
			date < 1939.1.1
		}
		country_event = {
			id = start_events.2
		}
	}
	else_if = {
		limit = {
			OR = {
				tag = CHI
				is_warlord = yes
			}
		}
		country_event = {

		}
	}
	else_if = {

				tag = JAP
				is_subject_of = JAP

	}
	else_if = {

	}
	else_if = {

		}
