# One weekly counter equals seven days.  Two bad checks create isolation;
# six bad checks create a blockade.  Two good checks provide recovery grace.
JAP_pacific_update_philippines_isolation = {

	else = { set_variable = { JAP_pacific_philippines_risk = 0 } add_to_variable = { JAP_pacific_philippines_recovery = 1 } }
	if = { limit = { check_variable = { JAP_pacific_philippines_risk > 1 } NOT = { has_country_flag = JAP_pacific_philippines_blockaded } } set_country_flag = JAP_pacific_philippines_isolated }
	if = { limit = { check_variable = { JAP_pacific_philippines_risk > 5 } } clr_country_flag = JAP_pacific_philippines_isolated set_country_flag = JAP_pacific_philippines_blockaded }
	if = { limit = { check_variable = { JAP_pacific_philippines_recovery > 1 } } clr_country_flag = JAP_pacific_philippines_isolated clr_country_flag = JAP_pacific_philippines_blockaded }
}
