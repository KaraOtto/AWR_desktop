CHI_power_blocs_initialize = {	#初始化民国影响力
	set_country_flag = CHI_power_blocs_initialized
	set_variable = { 
		CHI_power_bloc_leader_influence = 0.35 
	}
	set_variable = { 
		CHI_power_bloc_army_influence = 0.30 
	}
	set_variable = { 
		CHI_power_bloc_capital_influence = 0.20 
	}
	set_variable = { 
		CHI_power_bloc_technocrat_influence = 0.15 
	}
	CHI_power_blocs_normalize_and_refresh = yes
}

CHI_power_blocs_normalize = {
	clamp_variable = { 
		var = CHI_power_bloc_leader_influence 
		min = 0 
		max = 1 
	}
	clamp_variable = { 
		var = CHI_power_bloc_army_influence 
		min = 0 
		max = 1 
	}
	clamp_variable = { 
		var = CHI_power_bloc_capital_influence 
		min = 0 
		max = 1 
	}
	clamp_variable = { 
		var = CHI_power_bloc_technocrat_influence 
		min = 0 
		max = 1 
	}

	set_temp_variable = { 
		CHI_power_bloc_total = CHI_power_bloc_leader_influence 
	}
	add_to_temp_variable = { 
		CHI_power_bloc_total = CHI_power_bloc_army_influence 
	}
	add_to_temp_variable = { 
		CHI_power_bloc_total = CHI_power_bloc_capital_influence 
	}
	add_to_temp_variable = { 
		CHI_power_bloc_total = CHI_power_bloc_technocrat_influence 
	}

	if = {
		limit = { 
			check_variable = { 
				CHI_power_bloc_total < 0.001 
			} 
		}
		set_variable = { 
			CHI_power_bloc_leader_influence = 0.35 
		}
		set_variable = { 
			CHI_power_bloc_army_influence = 0.30 
		}
		set_variable = { 
			CHI_power_bloc_capital_influence = 0.20 
		}
		set_variable = { 
			CHI_power_bloc_technocrat_influence = 0.15 
		}
	}
	else = {
		divide_variable = { 
			var = CHI_power_bloc_leader_influence 
			value = CHI_power_bloc_total 
		}
		divide_variable = { 
			var = CHI_power_bloc_army_influence 
			value = CHI_power_bloc_total 
		}
		divide_variable = { 
			var = CHI_power_bloc_capital_influence 
			value = CHI_power_bloc_total 
		}
		divide_variable = { 
			var = CHI_power_bloc_technocrat_influence 
			value = CHI_power_bloc_total 
		}
	}
}

CHI_power_blocs_refresh_modifier = {
	if = {
		limit = { 
			has_dynamic_modifier = { 
				modifier = CHI_power_blocs_national_government 
			} 
		}
		remove_dynamic_modifier = { 
			modifier = CHI_power_blocs_national_government 
		}
	}
	if = {
		limit = { 
			has_dynamic_modifier = { 
				modifier = CHI_power_blocs_reformed_government 
			} 
		}
		remove_dynamic_modifier = { 
			modifier = CHI_power_blocs_reformed_government 
		}
	}
	if = {
		limit = { 
			has_country_flag = CHI_power_bloc_national_bourgeoisie 
		}
		add_dynamic_modifier = { 
			modifier = CHI_power_blocs_reformed_government 
		}
	}
	else = {
		add_dynamic_modifier = { 
			modifier = CHI_power_blocs_national_government 
		}
	}
	force_update_dynamic_modifier = yes
}

	clear_global_event_target = CHI_corruption_target
	clr_country_flag = CHI_corruption_group_army
	clr_country_flag = CHI_corruption_group_capital
	clr_country_flag = CHI_corruption_group_technocrat

	clr_country_flag = CHI_corruption_incident_air_kickback

CHI_corruption_monthly_tick = {
	add_to_variable = { 
		CHI_army_corruption = CHI_army_corruption_coefficient 
	}
	add_to_variable = { 
		CHI_capital_corruption = CHI_capital_corruption_coefficient 
	}
	add_to_variable = { 
		CHI_technocrat_corruption = CHI_technocrat_corruption_coefficient 
	}
	CHI_corruption_clamp = yes

	if = {
		limit = { 
			check_variable = { 
				CHI_corruption_monthly_priority < 0.5 
			} 
		}
		CHI_corruption_try_army = yes
		CHI_corruption_try_capital = yes
		CHI_corruption_try_technocrat = yes
		set_variable = { 
			CHI_corruption_monthly_priority = 1 
		}
	}
	else_if = {
		limit = { 
			check_variable = { 
				CHI_corruption_monthly_priority < 1.5 
			} 
		}
		CHI_corruption_try_capital = yes
		CHI_corruption_try_technocrat = yes
		CHI_corruption_try_army = yes
		set_variable = { 
			CHI_corruption_monthly_priority = 2 
		}
	}
	else = {
		CHI_corruption_try_technocrat = yes
		CHI_corruption_try_army = yes
		CHI_corruption_try_capital = yes
		set_variable = { 
			CHI_corruption_monthly_priority = 0 
		}
	}
}

CHI_corruption_try_army = {
	if = {
		limit = { 
			NOT = { 
				has_country_flag = CHI_corruption_event_pending 
			} 
		}
		randomize_temp_variable = { 
			var = CHI_corruption_army_roll 
			min = 0 
			max = 1 
		}
		if = { 
			limit = { 
				check_variable = { 
					CHI_corruption_army_roll < CHI_army_corruption 
				} 
			} 
			CHI_corruption_dispatch_army = yes 
		}
	}
}

CHI_corruption_apply_minor_incident = {

				has_country_flag = CHI_corruption_incident_land_arms 
			} 
			add_equipment_to_stockpile = { 
				type = infantry_equipment_1 
				amount = -1000 
			}

		}
		else_if = { 
			limit = { 
				has_country_flag = CHI_corruption_incident_land_ghost

		else_if = {

				has_country_flag = CHI_corruption_incident_navy_graft 
			} 
			add_timed_idea = { 
				idea = CHI_navy_corruption_graft_1 
				days = 180 
			} 
		}
		else_if = { 
			limit = {

				idea = CHI_navy_corruption_kickback_1

				idea = CHI_air_corruption_graft_1

		else = { 
			add_timed_idea = { 
				idea = CHI_air_corruption_kickback_1

	}
	else_if = {
