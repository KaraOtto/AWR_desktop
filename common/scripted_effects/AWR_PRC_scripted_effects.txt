PRC_support_up = {
	if = {
		limit = {
			NOT = {
				has_idea = PRC_communist_support_low
				has_idea = PRC_communist_support_med
				has_idea = PRC_communist_support_high
			}
		}
		add_ideas = PRC_communist_support_low
	}
	else_if = {
		limit = {
			has_idea = PRC_communist_support_low
		}
		swap_ideas = {
			remove_idea = PRC_communist_support_low
			add_idea = PRC_communist_support_med
		}
	}
	else_if = {
		limit = {
			has_idea = PRC_communist_support_med
		}
		swap_ideas = {
			remove_idea = PRC_communist_support_med
			add_idea = PRC_communist_support_high
		}
	}
	else_if = {
		limit = {
			has_idea = PRC_communist_support_high
		}
		set_global_flag = PRC_communist_support_maxed_out 
	}
}
PRC_support_down = {
	if = {
		limit = {
			has_idea = PRC_communist_support_high
			has_global_flag = PRC_communist_support_maxed_out
		}
		clr_global_flag = PRC_communist_support_maxed_out
	}
	else_if = {
		limit = {
			has_idea = PRC_communist_support_high
		}
		swap_ideas = {
			remove_idea = PRC_communist_support_high
			add_idea = PRC_communist_support_med
		}
	}
	else_if = {
		limit = {
			has_idea = PRC_communist_support_med
		}
		swap_ideas = {
			remove_idea = PRC_communist_support_med
			add_idea = PRC_communist_support_low
		}
	}
	else_if = {
		limit = {
			has_idea = PRC_communist_support_low
		}
		remove_ideas = PRC_communist_support_low
	}
}

modify_parliament_seat = {
	multiply_temp_variable = { 
		temp_seats = 200 
	} #times 2 for each 0.5 unit and times 100 for percentage
	subtract_from_temp_variable = { 
		temp_member = 1 
	}

	# check to see if result overflows
	set_temp_variable = { 
		temp = parliament_seat_array^temp_member 
	}
	add_to_temp_variable = { 
		temp = temp_seats 
	}

	if = {
		limit = {
			check_variable = { 
				temp > parliament_main_array^num 
			}
		}
		subtract_from_temp_variable = { 
			temp = parliament_main_array^num 
		}
		subtract_from_temp_variable = { 
			temp_seats = temp 
		}
	}
	else_if = {
		limit = {
			check_variable = { 
				temp < 0 
			}
		}
		subtract_from_temp_variable = { 
			temp_seats = temp 
		}
	}
	
	if = {
		limit = {
			check_variable = { 
				temp_seats < 0 
			} #substract only
			OR = {
				check_variable = { 
					parliament_seat_array^0 = 200 
				} #one party is completely dominant 
				check_variable = { 
					parliament_seat_array^1 = 200 
				} #one party is completely dominant 
				check_variable = { 
					parliament_seat_array^2 = 200 
				} #one party is completely dominant 
				check_variable = { 
					parliament_seat_array^3 = 200 
				} #one party is completely dominant
			}
		}
		if = {
			limit = {
				NOT = { 
					check_variable = { 
						parliament_seat_array^0 = 200 
					} 
				} #Not the dominant one
				has_country_leader = { 
					character = PRC_zhang_guotao 
					ruling_only = yes 
				} #has 0% minority leader
			}
			add_to_temp_array = { 
				temp_array = 0 
			} #zhang reemerge
		}
		else_if = {
			limit = {
				NOT = { 
					check_variable = { 
						parliament_seat_array^1 = 200 
					} 
				} #Not the dominant one
				OR = {
					has_country_leader = { 
						character = PRC_mao_zedong ruling_only = yes 
					} #has 0% minority leader
					has_country_leader = { 
						character = PRC_zhang_wentian 
						ruling_only = yes	
					} 
					has_country_leader = { 
						character = PRC_lin_boqu 
						ruling_only = yes	
					} 
				}
			}
			add_to_temp_array = { 
				temp_array = 1 
			} #mao reemerge
		}
		else_if = {
			limit = {
				NOT = { 
					check_variable = { 
						parliament_seat_array^2 = 200 
					} 
				} #Not the dominant one
				OR = {
					has_country_leader = { 
						character = prc_bo_gu 
						ruling_only = yes 
					} 
					has_country_leader = { 
						character = prc_wang_ming 
						ruling_only = yes	
					} 
				}
			}
			add_to_temp_array = { 
				temp_array = 2 
			} #28b reemerge
		}
		else_if = {
			limit = {
				NOT = { 
					check_variable = { 
						parliament_seat_array^3 = 200 
					} 
				} #Not the dominant one
				OR = {
					has_country_leader = { 
						character = prc_peng_shuzhi 
						ruling_only = yes 
					} #has 0% minority leader
					has_country_leader = { 
						character = PRC_chen_duxiu 
						ruling_only = yes	
					} 
				}
			}
			add_to_temp_array = { 
				temp_array = 3 
			} #trot reemerge
		}
		else_if = {
			limit = {
				NOT = { 
					check_variable = { 
						temp_member = 1 
					} 
				} #is not maoist
			}
			add_to_temp_array = { 
				temp_array = 1 
			} #maoist becomes the re-emerging party
		}
		else_if = {
			limit = {
				check_variable = { 
					temp_member = 1 
				} #is maoist
			}
			add_to_temp_array = { 
				temp_array = 2 
			} #28b becomes the re-emerging party
		}
	}
	else = {
		for_loop_effect = {
			start = 0
			end = parliament_seat_array^num
			if = {
				limit = {
					check_variable = { 
						parliament_seat_array^v > 0 
					} #zero support groups don't reappear 
				}
				add_to_temp_array = { 
					temp_array = v 
				}
			}			
		}
		# remove from list of parties the party to be added/subtracted
		remove_from_temp_array = { 
			array = temp_array value = temp_member 
		}
	}
		

	#the actual code to add/subtract seats
	add_to_variable = { 
		parliament_seat_array^temp_member = temp_seats 
	}


	#make sure it doesn't go above 200(max) or below zero

	# if its add
	if = {
		limit = {
			check_variable = { 
				temp_seats > 0 
			}
		}
		# loop until seats to be added is 0
		while_loop_effect = {
			limit = {
				check_variable = { 
					temp_seats > 0 
				}
			}

			# loops through remaining parties, subtracting 1 seat each time
			for_loop_effect = {
				start = 0
				end = temp_array^num

				if = {
					limit = {
						check_variable = { 
							parliament_seat_array^temp_array^v > 0 
						} # party must have more than 0 seats to be subtracted
						check_variable = { 
							temp_seats > 0 
						} # seats to be added is bigger than 0 currently
					}
					subtract_from_variable = { 
						parliament_seat_array^temp_array^v = 1 
					} # subtract from party 1 seat
					subtract_from_temp_variable = { 
						temp_seats = 1 
					} # subtract from seats to be added
				}
			}
		}
		
	}
	# if its subtract
	else_if = {
		limit = {
			check_variable = { 
				temp_seats < 0 
			}
			#check_variable = { parliament_seat_array^temp_member < 0 }
		}
		# loop until seats to be subtracted is 0
		while_loop_effect = {
			limit = {
				check_variable = { 
					temp_seats < 0 
				}
			}
			# loops through remaining parties, adding 1 seat each time
			for_loop_effect = {
				start = 0
				end = temp_array^num
				if = {
					limit = {
						check_variable = { 
							parliament_seat_array^temp_array^v < parliament_main_array^num 
						} # party must have less than max seats to be added
						check_variable = { 
							temp_seats < 0 
						} # seats to be subtracted is less than 0 currently
					}
					add_to_variable = { 
						parliament_seat_array^temp_array^v = 1 
					} # add to party 1 seat
					add_to_temp_variable = { 
						temp_seats = 1 
					} # add to seats to be subtracted
				}
			}
		}
	}
	# do nothing if 0
	# give party new seat value
	
	
	# Modify what shows up in the gui
	reload_parliament_gui = yes
}

X_xchinese_pick_propaganda_campaign_slot_effect = {
	hidden_effect = {
		if = {
			limit = {
				check_variable = { 
					X_xchinese_poster_left_icon_frame = X_xchinese_propaganda_empty_slot_index 
				}
			}
			set_variable = { 
				X_xchinese_poster_left_icon_frame = X_xchinese_new_propaganda_campaign 
			}
			#Ageing posters
			set_variable = { 
				X_xchinese_propaganda_left_poster_to_age = X_xchinese_poster_left_icon_frame 
			}
			hidden_effect = {
				country_event = { 
					id = xpand_xchinese_capmaign_events.101 
					days = X_xchinese_default_propaganda_campaign_old_days 
				}
				
				if = { 
					limit = {
						has_country_flag = X_xchinese_debug_fast_propaganda_campaigns_flag
					}
					set_country_flag = { 
						flag = X_xchinese_propaganda_left_days_remaining_flag 
						days = 5 
					}
				}
				else = { 
					set_country_flag = { 
						flag = X_xchinese_propaganda_left_days_remaining_flag 
						days = 180 
					}
				}
			}
		}
		else_if = {
			limit = {
				check_variable = { 
					X_xchinese_poster_center_icon_frame = X_xchinese_propaganda_empty_slot_index 
				}
			}
			set_variable = { 
				X_xchinese_poster_center_icon_frame = X_xchinese_new_propaganda_campaign 
			}
			#Ageing posters
			set_variable = { 
				X_xchinese_propaganda_center_poster_to_age = X_xchinese_poster_center_icon_frame 
			} 
			hidden_effect = {
				country_event = { 
					id = xpand_xchinese_capmaign_events.102 
					days = X_xchinese_default_propaganda_campaign_old_days 
				}
				
				if = { 
					limit = {
						has_country_flag = X_xchinese_debug_fast_propaganda_campaigns_flag
					}
					set_country_flag = { 
						flag = X_xchinese_propaganda_center_days_remaining_flag 
						days = 5 
					}
				}
				else = { 
					set_country_flag = { 
						flag = X_xchinese_propaganda_center_days_remaining_flag 
						days = 180 
					} 
				}
			}
		}
		else_if = {
			limit = {
				check_variable = { X_xchinese_poster_right_icon_frame = X_xchinese_propaganda_empty_slot_index }
			}
			set_variable = { X_xchinese_poster_right_icon_frame = X_xchinese_new_propaganda_campaign }
			#Ageing posters
			set_variable = { X_xchinese_propaganda_right_poster_to_age = X_xchinese_poster_right_icon_frame } #This is used to prevent the system to break if a propaganda campaign is cancelled
			hidden_effect = {
				country_event = { 
					id = xpand_xchinese_capmaign_events.103 
					days = X_xchinese_default_propaganda_campaign_old_days 
				}
				
				if = { #Debug -> Faster time
					limit = {
						has_country_flag = X_xchinese_debug_fast_propaganda_campaigns_flag
					}
					set_country_flag = { 
						flag = X_xchinese_propaganda_right_days_remaining_flag 
						days = 5 
					}
				}
				else = { #Regular time
					set_country_flag = { 
						flag = X_xchinese_propaganda_right_days_remaining_flag days = 180 
					} 
					
				}
			}
		}
		set_variable = { 
			X_xchinese_new_propaganda_campaign = 0 
		} # Reset var
		add_to_variable = { 
			X_xchinese_propaganda_campaigns_taken = 1 
		} 
	}
}

PRC_setup_third_world_independence_targets = {


	#########################################
	# 首先清空上一次扫描得到的名单
	#########################################

	clear_array = PRC_third_world_independence_targets



	#########################################
	# 第一部分：
	#
	# 搜索已经存在的殖民附属国
	#
	# 例如：
	# RAJ
	# INS
	# MAL
	# 等
	#########################################

	every_country = {

		limit = {

			exists = yes

			is_subject = yes


			# 排除中华人民共和国自己
			NOT = {
				original_tag = PRC
			}


			# 排除中华人民共和国自己的附属国
			NOT = {
				is_subject_of = PRC
			}


			################################
			# 首都必须处于我们的
			# 世界民族解放活动范围
			################################

			capital_scope = {

				OR = {

					is_on_continent = asia

					is_on_continent = middle_east

					is_on_continent = africa

					is_on_continent = south_america

					is_on_continent = australia
				}
			}
		}


		################################
		# 防止重复加入数组
		################################

		if = {

			limit = {

				PRC = {

					NOT = {

						is_in_array = {

							array = PRC_third_world_independence_targets

							value = PREV
						}
					}
				}
			}


			################################
			# 加入PRC目标数组
			################################

			PRC = {

				add_to_array = {

					array = PRC_third_world_independence_targets

					value = PREV
				}
			}
		}
	}



	#########################################
	# 第二部分：
	#
	# 搜索尚未存在、
	# 但核心领土处于殖民统治下的民族
	#
	# 例如：
	# KEN
	# NGA
	# 等
	#########################################

	every_state = {

		limit = {


			################################
			# 只检查我们的目标地区
			################################

			OR = {

				is_on_continent = asia

				is_on_continent = middle_east

				is_on_continent = africa

				is_on_continent = south_america

				is_on_continent = australia
			}


			################################
			# 这个州必须不是拥有者自己的核心
			#
			# 这是判断殖民统治的重要条件
			################################

			NOT = {
				is_core_of = owner
			}


			################################
			# 排除PRC自己的领土
			################################

			owner = {

				NOT = {
					original_tag = PRC
				}
			}


			################################
			# 必须存在至少一个核心国家
			################################

			check_variable = {
				core_countries^num > 0
			}
		}



		################################
		# 遍历这个州所有拥有核心的国家
		################################

		for_each_scope_loop = {

			array = core_countries


			################################
			# 此时：
			#
			# THIS = 拥有该州核心的国家
			################################

			if = {

				limit = {


					################################
					# 专门寻找尚未存在的国家
					################################

					exists = no


					################################
					# 排除PRC
					################################

					NOT = {
						original_tag = PRC
					}


					################################
					# 防止重复添加
					################################

					PRC = {

						NOT = {

							is_in_array = {

								array = PRC_third_world_independence_targets

								value = PREV
							}
						}
					}
				}


				################################
				# 加入世界民族解放目标池
				################################

				PRC = {

					add_to_array = {

						array = PRC_third_world_independence_targets

						value = PREV
					}
				}
			}
		}
	}

	# 第一种：
	#
	# 这个国家已经存在
	#
	# 给它留下：
	# “曾受PRC支持”的国家flag
	#########################################

	if = {

		limit = {
			exists = yes
		}

		set_country_flag = PRC_supported_independence_movement
	}
	#########################################
	# 第二种：
	#
	# 已经存在，而且是殖民附属国
	#
	# 主要通过自治度推动独立
	#########################################
	if = {

		limit = {

			exists = yes

			is_subject = yes

			NOT = {
				is_subject_of = PRC
			}
		}
		################################
		# 增加5%的自治进度
		################################
		add_autonomy_ratio = {

			value = 0.10

			localization = PRC_supported_national_liberation_autonomy
		}
	}
	#########################################
	# 第三种：
	# 遍历该民族所有核心州
	#
	# 对仍处于外国非核心统治下的地区
	# 支持地下民族运动
	#########################################
	every_core_state = {

		limit = {
			# 必须是殖民统治 / 非核心统治
			NOT = {
				is_core_of = owner
			}

			# PRC自己控制的地方不搞革命
			owner = {

				NOT = {
					original_tag = PRC
				}
			}
		}
		# 在州上留下永久记录
		# 即使以后这个国家通过起义出现，
		# state flag依然可以用来判断
		# PRC曾经支持过这里
		set_state_flag = PRC_supported_revolution_state

		# 一年内增加抵抗目标
		add_resistance_target = {

			amount = 8

			days = 365

			tooltip = PRC_third_world_resistance_target_tt
		}
		# 如果这个州已经启用了Resistance
		# 再立即提高5点
		if = {

			limit = {
				has_resistance = yes
			}

			add_resistance = 5
		}
	}
}

PRC_silver_dollar_war_daily_tick = {	#银元之战

    # 银元之战持续天数
    add_to_variable = {
        PRC_silver_dollar_war_days = 1
    }

    # ==========================================
    # 1. 投机活动自然推动银元上涨
    # ==========================================

    if = {
        limit = {
            check_variable = {
                var = RMB_belief
                value = 0.30
                compare = less_than
            }
        }

        # 人民币信用很低时，银元需求增加
        multiply_variable = {
            RMB_silver_dollar = 1.04
        }
    }

    else_if = {
        limit = {
            check_variable = {
                var = RMB_belief
                value = 0.50
                compare = less_than
            }
        }

        multiply_variable = {
            RMB_silver_dollar = 1.02
        }
    }

    # ==========================================
    # 2. 银元上涨反过来打击人民币信用
    # ==========================================

    if = {
        limit = {
            check_variable = {
                var = RMB_silver_dollar
                value = 1000
                compare = greater_than
            }
        }

        add_to_variable = {
            RMB_belief = -0.01
        }
    }

    if = {
        limit = {
            check_variable = {
                var = RMB_silver_dollar
                value = 1500
                compare = greater_than
            }
        }

        add_to_variable = {
            RMB_belief = -0.015
        }
    }

    # ==========================================
    # 3. 防止变量越界
    # ==========================================

    clamp_variable = {
        var = RMB_belief
        min = 0
        max = 1
    }

    clamp_variable = {
        var = RMB_silver_dollar
        min = 300
        max = 4000
    }

    # ==========================================
    # 4. 判断经济手段是否获胜
    # 人民币信用 >= 0.55
    # 银元汇率 <= 800
    # ==========================================

    if = {
        limit = {

            check_variable = {
                var = RMB_belief
                value = 0.55
                compare = greater_than_or_equals
            }

            check_variable = {
                var = RMB_silver_dollar
                value = 800
                compare = less_than_or_equals
            }
        }

        add_to_variable = {
            PRC_silver_dollar_stable_days = 1
        }
    }

    else = {
        set_variable = {
            var = PRC_silver_dollar_stable_days
            value = 0
        }
    }

    # 连续稳定3天 → 银元之战胜利
    if = {
        limit = {
            check_variable = {
                var = PRC_silver_dollar_stable_days
                value = 3
                compare = greater_than_or_equals
            }
        }

        clr_country_flag = silver_dollar_war_start
        set_country_flag = silver_dollar_war_economic_victory

        country_event = {
            id = AWR_PRC.275
            hours = 6
        }
    }
}

PRC_g_postwar_transfer = { #抗战结束后的支持地区移交
	if = {
		limit = {
			has_country_flag = PRC_g_fought_japan
			NOT = {
				has_country_flag = PRC_g_postwar_transferred
			}
			NOT = {
				has_war_with = JAP
			}
			OR = {
				has_global_flag = CHI_no_longer_united_flag
				JAP = {
					has_capitulated = yes
				}
				NOT = {
					country_exists = JAP
				}
				CHI = {
					NOT = {
						has_war_with = JAP
					}
				}
			}
		}
		every_state = {
			limit = {
				has_dynamic_modifier = {
					modifier = PRC_g_support
				}
				PRC_g_china_state = yes #中国地区范围
				NOT = {
					PRC_g_state_blocked = yes #地区被反制措施封锁
				}
				controller = {
					NOT = {
						PRC_g_is_japanese_side = yes #日本或其附属及盟友
					}
				}
			}
			add_core_of = PRC
			PRC = {
				transfer_state = PREV
			}
		}
		set_country_flag = PRC_g_postwar_transferred
		country_event = { #敌后根据地归属确定
			id = PRC_guerrilla.4
		}
	}
}

PRC_g_refresh_map_query = { #刷新所查询地区的当前状态
	if = {
		limit = {
			PRC_g_can_view_map = yes #失去查看权限时停止刷新查询
			has_country_flag = PRC_g_query_open
			has_variable = PRC_g_selected_state
		}
		set_variable = {
			PRC_g_query_activity = 0 #查询：游击队活动
		}
		set_variable = {
			PRC_g_query_support = -1 #查询：共产党群众支持；-1 表示无权查看

		}

					PRC_g_query_controller = PREV.controller #查询：地区控制者

						has_dynamic_modifier = {

						}
					}
					PREV = {

PRC_g_spawn_regional_divisions = { #按地区编制生成全部当地游击队师，随后清零
	if = {
		limit = {
			check_variable = {
				PRC_g_divisions > 0 #当地游击队师数量
			}
		}
		if = {
			limit = {
				OR = {
					region = 242
					region = 243
					region = 155
				}
			}
			PRC_g_spawn_army_242 = yes #生成东北野战军
		}
		else_if = {
			limit = {
				OR = {
					region = 244
					region = 143
				}
			}
			PRC_g_spawn_army_244 = yes #生成华北野战军
		}
		else_if = {
			limit = {
				OR = {
					region = 245
					region = 144
					region = 200
					region = 250
				}
			}
			PRC_g_spawn_army_245 = yes #生成第一野战军
		}
		else_if = {
			limit = {
				OR = {
					region = 247
					region = 165
					region = 246
				}
			}
			PRC_g_spawn_army_247 = yes #生成中原野战军
		}
		else_if = {
			limit = {
				OR = {
					region = 305
					region = 164
				}
			}
			PRC_g_spawn_army_305 = yes #生成华东野战军
		}
		else_if = {
			limit = {
				OR = {
					region = 249
					region = 248
				}
			}
			PRC_g_spawn_army_249 = yes #生成华南野战军
		}
		else = {
			PRC_g_spawn_army_244 = yes #生成华北野战军
		}
		PRC = {
			set_temp_variable = {
				PRC_g_rifles = PREV.PRC_g_divisions
			}
			multiply_temp_variable = {
				PRC_g_rifles = 750
			}
			set_temp_variable = {
				PRC_g_artillery = PREV.PRC_g_divisions
			}
			multiply_temp_variable = {
				PRC_g_artillery = 9
			}
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = var:PRC_g_rifles
			}
			add_equipment_to_stockpile = {
				type = artillery_equipment
				amount = var:PRC_g_artillery
				producer = JAP
			}
		}
		set_variable = {
			PRC_g_divisions = 0 #当地游击队师数量
		}
	}
}

PRC_g_reorganize_region_242 = { #汇总并整编北满洲游击队
	if = {
		limit = {
			any_owned_state = {
				region = 242
				is_fully_controlled_by = ROOT
			}
		}
		set_variable = {
			PRC_g_assembly_total = 0 #整编区域内的游击队师总数
		}
		every_state = {
			limit = {
				region = 242
				check_variable = {
					PRC_g_divisions > 0 #当地游击队师数量
				}
			}
			PRC = {
				add_to_variable = {
					PRC_g_assembly_total = PREV.PRC_g_divisions #整编区域内的游击队师总数
				}
			}
		}
		if = {
			limit = {
				check_variable = {
					PRC_g_assembly_total > 0 #整编区域内的游击队师总数
				}
			}
			every_state = {
				limit = {
					region = 242
				}
				set_variable = {
					PRC_g_divisions = 0 #当地游击队师数量
				}
			}
			random_owned_controlled_state = {
				limit = {
					region = 242
				}
				prioritize = {
					597
					615
					749
				}
				set_variable = {
					PRC_g_divisions = PRC.PRC_g_assembly_total #当地游击队师数量
				}
				PRC_g_spawn_regional_divisions = yes #按地区编制生成全部当地游击队师，随后清零
			}
		}
		clear_variable = PRC_g_assembly_total
	}
}
