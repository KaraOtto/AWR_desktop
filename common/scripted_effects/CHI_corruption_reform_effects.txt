# Reform progress supplements the existing corruption stock and monthly modifiers.
CHI_corruption_integrity_clamp = {
	clamp_variable = {
		var = Institutional_Integrity
		min = 0
		max = 100
	}
}

CHI_corruption_reform_initialize = {

		limit = {
			NOT = {

			}
		}
		set_country_flag = CHI_corruption_reform_initialized
		set_variable = {
			Institutional_Integrity = 0
		}
	}
}

CHI_corruption_reform_daily = {
	if = {
		limit = {
			OR = {
				has_country_flag = CHI_corruption_central_campaign
				has_country_flag = CHI_corruption_communist_campaign
			}
		}
		add_to_variable = {
			CHI_corruption_campaign_elapsed = 1
		}
		if = {
			limit = {
				NOT = {
					has_country_flag = CHI_corruption_event_pending
				}
				check_variable = {
					var = CHI_corruption_campaign_elapsed
					value = CHI_corruption_campaign_next
					compare = greater_than_or_equals
				}
				check_variable = {
					CHI_corruption_campaign_issued < CHI_corruption_campaign_total
				}
			}
			# Round-robin selection gives the communist campaign two cases per group.
			if = {
				limit = {
					check_variable = {
						CHI_corruption_campaign_group < 0.5
					}
				}
				CHI_corruption_prepare_army = yes
				set_variable = {
					CHI_corruption_campaign_group = 1
				}
			}
			else_if = {
				limit = {
					check_variable = {
						CHI_corruption_campaign_group < 1.5
					}
				}
				CHI_corruption_prepare_capital = yes
				set_variable = {
					CHI_corruption_campaign_group = 2
				}
			}
			else = {
				CHI_corruption_prepare_technocrat = yes
				set_variable = {
					CHI_corruption_campaign_group = 0
				}
			}
			set_country_flag = CHI_corruption_campaign_case
			add_to_variable = {
				CHI_corruption_campaign_issued = 1
			}
			add_to_variable = {
				CHI_corruption_campaign_next = CHI_corruption_campaign_interval
			}
			country_event = {
				id = CHI_corruption.30
			}
		}
		if = {
			limit = {
				check_variable = {
					CHI_corruption_campaign_issued > CHI_corruption_campaign_total
				}
			}
			# Defensive clamp for edited saves; normal completion is equality below.
			set_variable = {
				CHI_corruption_campaign_issued = CHI_corruption_campaign_total
			}
		}
		if = {
			limit = {
				check_variable = {
					CHI_corruption_campaign_issued = CHI_corruption_campaign_total
				}
			}
			clr_country_flag = CHI_corruption_central_campaign
			clr_country_flag = CHI_corruption_communist_campaign
		}
	}
	if = {
		limit = {
			NOT = {
				has_country_flag = CHI_corruption_event_pending
			}
		}
		if = {
			limit = {
				check_variable = {
					CHI_corruption_pending_severe > 0
				}
			}
			add_to_variable = {
				CHI_corruption_pending_severe = -1
			}
			CHI_corruption_prepare_random = yes
			country_event = {
				id = CHI_corruption.30
			}
		}
		else_if = {
			limit = {
				check_variable = {
					CHI_corruption_pending_medium > 0
				}
			}
			add_to_variable = {
				CHI_corruption_pending_medium = -1
			}
			CHI_corruption_prepare_random = yes
			country_event = {
				id = CHI_corruption.20
			}
		}
		else_if = {
			limit = {
				check_variable = {
					CHI_corruption_pending_minor > 0
				}
			}
			add_to_variable = {
				CHI_corruption_pending_minor = -1
			}
			CHI_corruption_prepare_random = yes
			country_event = {
				id = CHI_corruption.10
			}
		}
		else_if = {
			limit = {
				check_variable = {
					CHI_corruption_audit_army_pending > 0
				}
			}
			add_to_variable = {
				CHI_corruption_audit_army_pending = -1
			}
			CHI_corruption_dispatch_army = yes
		}
		else_if = {
			limit = {
				check_variable = {
					CHI_corruption_audit_capital_pending > 0
				}
			}
			add_to_variable = {
				CHI_corruption_audit_capital_pending = -1
			}
			CHI_corruption_dispatch_capital = yes
		}
		else_if = {
			limit = {
				check_variable = {
					CHI_corruption_audit_technocrat_pending > 0
				}
			}
			add_to_variable = {
				CHI_corruption_audit_technocrat_pending = -1
			}
			CHI_corruption_dispatch_technocrat = yes
		}
	}
}

CHI_corruption_educate_one = {
	random_character = {
		limit = {
			has_character_flag = CHI_corruption_education_candidate
			NOT = {
				has_character_flag = CHI_corruption_educated_this_session
			}
		}
		set_character_flag = CHI_corruption_educated_this_session
		if = {
			limit = {
				CHI_has_corruption_trait = no
			}
			set_character_flag = CHI_corruption_whitelist
			if = {
				limit = {
					has_character_flag = CHI_corruption_education_military
				}
				add_unit_leader_trait = CHI_corruption_whitelist
			}
			else = {
				add_trait = {
					slot = political_advisor
					trait = CHI_corruption_whitelist_advisor
				}
			}
		}
		if = {
			limit = {
				has_character_flag = CHI_corruption_education_military
			}
			CHI_corruption_clean_unit_leader = yes
		}
		if = {
			limit = {
				has_advisor_role = political_advisor
			}
			CHI_corruption_clean_advisor = yes
		}
	}
}

# Secret investigation uses its own target, independent of ordinary corruption cases.
CHI_corruption_secret_investigation = {
	every_character = {
		limit = {
			CHI_corruption_secret_target_eligible = yes
		}
		set_character_flag = CHI_corruption_investigation_candidate
	}
	if = {
		limit = {
			has_country_flag = CHI_corruption_investigating_capital
			any_character = {
				has_character_flag = CHI_corruption_investigation_candidate
				has_trait = capital_corruption
			}
		}
		every_character = {
			limit = {
				NOT = {
					has_trait = capital_corruption
				}
			}
			clr_character_flag = CHI_corruption_investigation_candidate
		}
	}
	else_if = {
		limit = {
			NOT = {
				has_country_flag = CHI_corruption_investigating_capital
			}
			any_character = {
				has_character_flag = CHI_corruption_investigation_candidate
				has_trait = technocrat_corruption
			}
		}
		every_character = {
			limit = {
				NOT = {
					has_trait = technocrat_corruption
				}
			}
			clr_character_flag = CHI_corruption_investigation_candidate
		}
	}
	else_if = {
		limit = {
			any_character = {
				has_character_flag = CHI_corruption_investigation_candidate
				CHI_has_corruption_trait = yes
			}
		}
		every_character = {
			limit = {
				CHI_has_corruption_trait = no
			}
			clr_character_flag = CHI_corruption_investigation_candidate
		}
	}
	random_character = {
		limit = {
			has_character_flag = CHI_corruption_investigation_candidate
		}
		save_global_event_target_as = CHI_corruption_investigation_target
	}
	every_character = {
		clr_character_flag = CHI_corruption_investigation_candidate
	}
	set_country_flag = CHI_corruption_secret_event_pending
	set_country_flag = CHI_corruption_event_pending
	set_country_flag = {
		flag = CHI_corruption_secret_cooldown
		days = 30
	}
	country_event = {
		id = CHI_corruption_reform.1
	}
}
